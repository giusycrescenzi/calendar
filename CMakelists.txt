cmake_minimum_required(VERSION 3.16)
project(agenda LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Trova le dipendenze

find_package(Qt6 COMPONENTS Widgets REQUIRED)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(CMAKE_AUTOUIC ON)

# Abilita i test
enable_testing()

# Trova GTest
find_package(GTest REQUIRED)

# Definizione della Libreria della Logica
add_library(corelib
    src/core/Day.cpp
    src/core/Register.cpp
    src/core/Task.cpp
)
target_include_directories(corelib PUBLIC
    ${CMAKE_CURRENT_SOURCE_DIR}/src/core
)

# Eseguibile dell'Applicazione Principale
add_executable(agenda
    src/main.cpp
)
target_link_libraries(agenda PRIVATE Qt6::Widgets corelib)

# Eseguibile dei Test

add_executable(runCoreTests
    test/test_main.cpp
    test/test_core.cpp
)

# Collega i test a GTest e a corelib.
target_link_libraries(runCoreTests PRIVATE GTest::gtest_main corelib)

# Scopri i test
include(GoogleTest)
gtest_discover_tests(runCoreTests)
